# Commit Scopes for go-convert
# Use these scopes in conventional commits: type(scope): description
# The scope is optional: omit it for repository-wide changes. Never use `root`.
# Only the first word of a non-comment line is a scope.
# Example: feat(proto): convert oneof fields

# Repository scopes
deps                        # Dependency updates (Dependabot writes build(deps): bump ...)
README                      # Root README
docs                        # Documentation

# Packages
convert                     # Root package: converter facade, options and errors
codec                       # Codec contract and registry
codec/mapslice              # Map and slice codecs
codec/unixtime              # Unix time codecs
codec/optionalcodec         # optional value codecs
codec/textcodec             # TextMarshaler and TextUnmarshaler codecs
codec/enumcodec             # Enum and string or integer codecs
codec/sqlnull               # database/sql null type codecs
codec/durationcodec         # Duration string and number codecs
codec/b64codec              # Base64 byte slice codecs
codec/jsoncodec             # JSON value codecs
converttest                 # Test helpers for consumers
internal/engine             # Conversion engine
internal/primitive          # Primitive kind conversions
internal/typeinfo           # Type metadata cache
internal/reflect            # Reflection helpers
proto                       # Protobuf conversion helpers, nested module
proto/codec/jsonpb          # Protobuf JSON codecs
proto/codec/tspb            # Timestamp codecs
proto/codec/durpb           # Duration codecs
proto/codec/pbwrap          # Wrapper type codecs
proto/codec/oneof           # Oneof codecs
proto/codec/enumpb          # Protobuf enum codecs
proto/codec/fieldmaskpb     # Field mask codecs
proto/codec/anypb           # Any message codecs
googletype                  # google.type codecs, nested module

# Tooling
devtools                    # Development tool dependencies
ci                          # GitHub Actions workflows and scripts
